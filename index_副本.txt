        // Smooth Scrolling for Navigation
        document.querySelectorAll('a[href^="#"]').forEach(anchor => {
            anchor.addEventListener('click', function (e) {
                e.preventDefault();
                const targetId = this.getAttribute('href');
                if(targetId === '#') return;
                const targetElement = document.querySelector(targetId);
                if(targetElement){
                    targetElement.scrollIntoView({
                        behavior: 'smooth'
                    });
                }
            });
        });

        // Function to scroll to specific theme (for Recommended Movies click)
        function scrollToTheme(themeId) {
            const element = document.getElementById(themeId);
            if(element) {
                element.scrollIntoView({ behavior: 'smooth', block: 'center' });
            }
        }

        // Modal Logic
        const modalOverlay = document.getElementById('modalOverlay');
        const modalTitle = document.getElementById('modalTitle');
        const modalDesc = document.getElementById('modalDesc');
        const modalImg = document.getElementById('modalImg');

        // Function to open modal with specific content
        function openModal(title, desc, imgSrc) {
            modalTitle.innerText = title;
            modalDesc.innerHTML = desc; // Use innerHTML to allow <br> tags
            modalImg.src = imgSrc;
            
            modalOverlay.classList.add('active');
            // Prevent body scroll when modal is open
            document.body.style.overflow = 'hidden';
        }

        // Close when clicking outside content (on the overlay)
        function closeModal(e) {
            if (e.target === modalOverlay) {
                closeModalDirect();
            }
        }

        // Close when clicking the X button
        function closeModalDirect() {
            modalOverlay.classList.remove('active');
            // Restore body scroll
            document.body.style.overflow = 'auto';
        }

        // Close on Escape key
        document.addEventListener('keydown', function(e) {
            if (e.key === "Escape") {
                closeModalDirect();
            }
        });
    </script>
